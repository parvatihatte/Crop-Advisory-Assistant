9. DATABASE SCHEMA (PRODUCTION SQL FOR POSTGRESQL)
Execute this DDL schema script using Drizzle ORM or native SQL execution:
SQL
-- Enable UUID extension
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

-- USER ACCOUNTS
CREATE TABLE IF NOT EXISTS users (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    email VARCHAR(255) UNIQUE NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    full_name VARCHAR(100) NOT NULL,
    role VARCHAR(50) NOT NULL DEFAULT 'farmer' CHECK (role IN ('farmer', 'agronomist', 'admin')),
    preferred_language VARCHAR(10) NOT NULL DEFAULT 'en',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- FARMS
CREATE TABLE IF NOT EXISTS farms (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    name VARCHAR(150) NOT NULL,
    location_name VARCHAR(255) NOT NULL,
    latitude NUMERIC(10, 8),
    longitude NUMERIC(11, 8),
    climate_zone VARCHAR(50) NOT NULL,
    total_area_hectares NUMERIC(8, 2) NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- FIELDS / PLOTS
CREATE TABLE IF NOT EXISTS fields (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    farm_id UUID NOT NULL REFERENCES farms(id) ON DELETE CASCADE,
    name VARCHAR(100) NOT NULL,
    area_hectares NUMERIC(8, 2) NOT NULL,
    soil_type VARCHAR(50) NOT NULL,
    soil_ph NUMERIC(3, 1) CHECK (soil_ph BETWEEN 3.0 AND 10.0),
    nitrogen_level VARCHAR(20) DEFAULT 'Medium',
    phosphorus_level VARCHAR(20) DEFAULT 'Medium',
    potassium_level VARCHAR(20) DEFAULT 'Medium',
    organic_matter_pct NUMERIC(4, 2),
    irrigation_type VARCHAR(50) NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- CROP ADVISORIES
CREATE TABLE IF NOT EXISTS advisories (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    field_id UUID NOT NULL REFERENCES fields(id) ON DELETE CASCADE,
    user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    advisory_type VARCHAR(50) NOT NULL CHECK (advisory_type IN ('CROP_PLANNING', 'SOIL_HEALTH', 'DISEASE_DIAGNOSIS', 'IRRIGATION')),
    target_crop VARCHAR(100) NOT NULL,
    growth_stage VARCHAR(50) NOT NULL,
    symptoms_description TEXT,
    image_url TEXT,
    
    -- Structured AI Output Stored as JSONB
    ai_raw_response JSONB NOT NULL,
    summary_title VARCHAR(255) NOT NULL,
    health_status VARCHAR(50) NOT NULL, -- e.g., Healthy, At Risk, Severe Issues
    confidence_score NUMERIC(3, 2),     -- 0.00 to 1.00
    
    language_code VARCHAR(10) NOT NULL DEFAULT 'en',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- INDEXES FOR FAST MULTI-TENANT QUERYING
CREATE INDEX IF NOT EXISTS idx_farms_user_id ON farms(user_id);
CREATE INDEX IF NOT EXISTS idx_fields_farm_id ON fields(farm_id);
CREATE INDEX IF NOT EXISTS idx_advisories_field_id ON advisories(field_id);
CREATE INDEX IF NOT EXISTS idx_advisories_user_id ON advisories(user_id);

Data Sanitization: Sanitize all textual user inputs using xss or backend escape utilities prior to storing or supplying context to the LLM.

    "preventative_measures"

export async function generateCropAdvisory(payload: any) {
  const prompt = `
Generate a comprehensive agricultural advisory based on the following field parameters:
- Target Crop: ${payload.targetCrop}
- Growth Stage: ${payload.growthStage}
- Soil Type: ${payload.soilType}
- Soil pH: ${payload.soilPh}
- Soil N-P-K Status: N=${payload.nitrogenLevel}, P=${payload.phosphorusLevel}, K=${payload.potassiumLevel}
- Irrigation Source: ${payload.irrigationType}
- Special Observed Symptoms/Notes: ${payload.symptomsDescription || "None reported"}
- Output Target Language Code: ${payload.languageCode || 'en'}

Return ONLY a JSON response following the exact schema provided.
`;

  const response = await ai.models.generateContent({
    model: GEMINI_MODEL_FLASH,
    contents: prompt,
    config: {
      responseMimeType: "application/json",
      responseSchema: cropAdvisorySchema,
      temperature: 0.2,
    }
  });

  return JSON.parse(response.text!);
}

      },
      required: ["common_name", "scientific_name", "severity", "primary_cause"]
    },
    symptoms_identified: {
      type: Type.ARRAY,
      items: { type: Type.STRING }
    },
    treatment_protocol: {
      type: Type.OBJECT,
      properties: {
        immediate_action: { type: Type.STRING },
        organic_controls: { type: Type.ARRAY, items: { type: Type.STRING } },
        chemical_controls: { type: Type.ARRAY, items: { type: Type.STRING } }
      },
      required: ["immediate_action", "organic_controls", "chemical_controls"]
    },
    preventative_guidance: {
      type: Type.ARRAY,
      items: { type: Type.STRING }
    }
  },
  required: [
    "summary_title",
    "health_status",
    "confidence_score",
    "detected_issue",
    "symptoms_identified",
    "treatment_protocol",
    "preventative_guidance"
  ]
};

export async function diagnoseCropImage(imageBase64: string, mimeType: string, context: any) {
  const prompt = `
Analyze this crop/plant leaf image for signs of disease, pest infestation, or nutrient stress.
Field Context:
- Crop Type: ${context.targetCrop}
- Growth Stage: ${context.growthStage}
- Observed Field Symptoms: ${context.symptomsDescription || "N/A"}
- Language: ${context.languageCode || "en"}

Identify any disease or condition with precision, provide immediate remediation, organic treatments, chemical controls (if appropriate), and preventative steps.
`;

  const imagePart = {
    inlineData: {
      data: imageBase64,
      mimeType: mimeType
    }
  };

  const response = await ai.models.generateContent({
    model: GEMINI_MODEL_FLASH,
    contents: [prompt, imagePart],
    config: {
      responseMimeType: "application/json",
      responseSchema: visualDiagnosticSchema,
      temperature: 0.1,
    }
  });

  return JSON.parse(response.text!);
}
